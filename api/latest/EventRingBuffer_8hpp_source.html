105 EventRingBuffer(void* memory, std::uint32_t count, std::uint32_t payloadSize)

108 , _payloadSize{payloadSize}
109 , _stride{slotStride(payloadSize)}

139 mxlStatus publish(std::uint64_t index, mxlEventInfo const& info, std::uint8_t const* payload) noexcept
140 {
141 if (index > MAX_INDEX)
142 {
143 return MXL_ERR_OUT_OF_RANGE_TOO_LATE;
144 }
145 auto& event = slot(index);
146 auto sequence = std::atomic_ref{event.header.sequence};
147 auto const previous = sequence.load(std::memory_order_relaxed);
148 if ((previous != EMPTY) && (previous != (index << 1)) && ((previous >> 1) >= index))
149 {
150 // An unpublished writing tag for this index is safe to retry after a crash.
151 // A completed tag must never be reused: readers may already have observed it.
152 return MXL_ERR_FLOW_INVALID;
153 }
154 sequence.store(index << 1, std::memory_order_release);
155 storeBytes(event.header.infoWords, &info, sizeof info);
156 storeBytes(payloadWords(event), payload, info.eventSize);
157 sequence.store((index << 1) | 1, std::memory_order_release);
158 return MXL_STATUS_OK;
159 }

169 ReadResult read(std::uint64_t index, mxlEventInfo& info, std::uint8_t* payload) const noexcept
170 {
171 if (index > MAX_INDEX)
172 {
173 return ReadResult::Pending;
174 }
175 auto& event = slot(index);
176 auto const expected = (index << 1) | 1;
177 auto const before = load(event.header.sequence);
178 if (before != expected)
179 {
180 return (before != EMPTY) && ((before >> 1) > index) ? ReadResult::Overwritten : ReadResult::Pending;
181 }
182 loadBytes(&info, event.header.infoWords, sizeof info);
183 // Metadata may be an inconsistent snapshot until the final sequence check.
184 if (info.eventSize <= _payloadSize)
185 {
186 loadBytes(payload, payloadWords(event), info.eventSize);
187 }
188 if (load(event.header.sequence) != before)
189 {
190 return ReadResult::Overwritten;
191 }
192 if ((info.version != EVENT_HEADER_VERSION) || (info.size != sizeof info) || (info.flags != 0) || (info.eventSize > _payloadSize))
193 {
194 return ReadResult::Invalid;
195 }
196 info.index = index; // Queue position is library-owned, not producer metadata.
197 return ReadResult::Ready;
198 }

205 static std::uint64_t load(std::uint64_t const& value) noexcept
206 {
207 // libc++ versions used by MXL do not support atomic_ref<const T>.
208 // This performs loads only, including on read-only IPC mappings.
209 return std::atomic_ref{const_cast<std::uint64_t&>(value)}.load(std::memory_order_acquire);
210 }

252 static void storeBytes(std::uint64_t* destination, void const* source, std::size_t size) noexcept
253 {
254 auto const bytes = static_cast<std::uint8_t const*>(source);
255 for (auto offset = std::size_t{0}; offset < size; offset += sizeof *destination)
256 {
257 auto word = std::uint64_t{};
258 std::memcpy(&word, bytes + offset, std::min(sizeof word, size - offset));
259 // If a reader observes any new word, this release/acquire pair
260 // forces its final sequence load to observe the writing tag or later.
261 std::atomic_ref{destination[offset / sizeof word]}.store(word, std::memory_order_release);
262 }
263 }
264
271 static void loadBytes(void* destination, std::uint64_t const* source, std::size_t size) noexcept
272 {
273 auto bytes = static_cast<std::uint8_t*>(destination);
274 for (auto offset = std::size_t{0}; offset < size; offset += sizeof *source)
275 {
276 auto const word = load(source[offset / sizeof(std::uint64_t)]);
277 std::memcpy(bytes + offset, &word, std::min(sizeof word, size - offset));
278 }
279 }
