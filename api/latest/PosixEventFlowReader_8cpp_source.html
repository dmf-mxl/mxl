99 mxlStatus PosixEventFlowReader::getEvent(Timepoint deadline, mxlEventInfo* info, std::uint8_t** payload)
100 {
101 auto flow = _flowData->flow();
102 auto sync = std::atomic_ref{flow->state.syncCounter};
103 while (true)
104 {
105 auto const previous = sync.load(std::memory_order_acquire);
106 auto const oldest = _flowData->oldestIndex();
107 if (_nextIndex < oldest)
108 {
109 _nextIndex = oldest;
110 return MXL_ERR_OUT_OF_RANGE_TOO_LATE;
111 }
112 auto received = mxlEventInfo{};
113 auto const result = _flowData->ring().read(_nextIndex, received, _snapshot.data());
114 if (result == EventRingBuffer::ReadResult::Ready)
115 {
116 ++_nextIndex;
117 *info = received;
118 *payload = _snapshot.data();
119 (void)updateFileAccessTime(_accessFileFd);
120 return MXL_STATUS_OK;
121 }
122 if (result == EventRingBuffer::ReadResult::Overwritten)
123 {
124 _nextIndex = std::max(_nextIndex + 1, _flowData->oldestIndex());
125 return MXL_ERR_OUT_OF_RANGE_TOO_LATE;
126 }
127 if (result == EventRingBuffer::ReadResult::Invalid)
128 {
129 return MXL_ERR_FLOW_INVALID;
130 }
131 if (!isFlowValidImpl())
132 {
133 return MXL_ERR_FLOW_INVALID;
134 }
135 if ((currentTime(Clock::Realtime) >= deadline) || !waitUntilChanged(&flow->state.syncCounter, previous, deadline))
136 {
137 return MXL_ERR_OUT_OF_RANGE_TOO_EARLY;
138 }
139 }
140 }
