13/*
14 This test verifies that a read request for the ring buffer's
15 tailIndex grain returns TOO_LATE when the writer has a write
16 operation open at headIndex+1.
17
18 +-----------+-- --+-------------+-----------+-------------+--
19 | index 0 | | index N-2 | index N-1 | index 0 |
20 | tailIndex | ... | headIndex-1 | headIndex | headIndex+1 | ...
21 +-----------+-- --+-------------+-----------+-------------+--
22 ^
23 ring wrap
24
25 N = grainCount
26
27 The diagram illustrates that tailIndex and headIndex+1 address the
28 same ring-buffer index.
29
30 Note: index 0 is used here for illustration. The actual ring-buffer
31 starting index is tailIndex % grainCount.
32*/
33TEST_CASE_PERSISTENT_FIXTURE(mxl::tests::mxlDomainFixture, "Video Flow : tail read during open write", "[mxl flows]")
34{
35 auto const flowId = "5fbec3b1-1b0f-417d-9059-8b94a47197ed";
36 auto const flowDef = mxl::tests::readFile("data/v210_flow.json");
37
38 // Init the domain instance.
39 auto instance = mxlCreateInstance(domain.string().c_str(), "");
40 REQUIRE(instance != nullptr);
41
42 // Init a flow writer.
43 auto writer = mxlFlowWriter{};
44 auto configInfo = mxlFlowConfigInfo{};
45 auto flowWasCreated = false;
46
47 REQUIRE(mxlCreateFlowWriter(instance, flowDef.c_str(), "", &writer, &configInfo, &flowWasCreated) == MXL_STATUS_OK);
48 REQUIRE(flowWasCreated);
49
50 // Init headIndex from the current MXL time.
51 auto const now = mxlGetTime();
52 auto headIndex = mxlTimestampToIndex(&configInfo.common.grainRate, now);
53 REQUIRE(headIndex != MXL_UNDEFINED_INDEX);
54
55 auto const grainCount = static_cast<std::uint64_t>(configInfo.discrete.grainCount);
56
57 REQUIRE(grainCount > 1U);
58 REQUIRE(headIndex > grainCount);
59
60 // Init each grain in the ring buffer.
61 auto tailIndex = headIndex - grainCount + 1;
62 for (auto i = std::uint64_t{0}; i < grainCount; ++i)
63 {
64 auto grainInfo = mxlGrainInfo{};
65 std::uint8_t* buffer = nullptr;
66
67 REQUIRE(mxlFlowWriterOpenGrain(writer, tailIndex + i, &grainInfo, &buffer) == MXL_STATUS_OK);
68
69 buffer[0] = 0xCA;
70 buffer[grainInfo.grainSize - 1] = 0xFE;
71 grainInfo.validSlices = grainInfo.totalSlices;
72
73 REQUIRE(mxlFlowWriterCommitGrain(writer, &grainInfo) == MXL_STATUS_OK);
74 }
75
76 // Init a reader and sanity check the flow's headIndex.
77 auto reader = mxlFlowReader{};
78 REQUIRE(mxlCreateFlowReader(instance, flowId, "", &reader) == MXL_STATUS_OK);
79
80 auto runtimeInfo = mxlFlowRuntimeInfo{};
81 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
82 REQUIRE(runtimeInfo.headIndex == headIndex);
83
84 // Reading the tailIndex grain should be TOO_LATE when write is
85 // not in progress.
86 auto readGrainInfo = mxlGrainInfo{};
87 std::uint8_t* readBuffer = nullptr;
88 auto const grainReadStatusOutsideWrite = mxlFlowReaderGetGrainNonBlocking(reader, tailIndex, &readGrainInfo, &readBuffer);
89 {
90 INFO("The tailIndex grain read should return TOO_LATE even when no write "
91 "is in progress because the tail grain is reserved for the writer.");
92 CHECK(grainReadStatusOutsideWrite == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
93 }
94
95 // Reading the tailIndex grain through the slice API should be
96 // TOO_LATE when write is not in progress.
97 readBuffer = nullptr;
98 readGrainInfo = {};
99 auto const sliceReadStatusOutsideWrite = mxlFlowReaderGetGrainSliceNonBlocking(
100 reader, tailIndex, MXL_GRAIN_VALID_SLICES_ANY, &readGrainInfo, &readBuffer);
101 {
102 INFO("The tailIndex slice read should return TOO_LATE even when no write "
103 "is in progress because the tail grain is reserved for the writer.");
104 CHECK(sliceReadStatusOutsideWrite == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
105 }
106
107 // Open a write to the next sequential grain at headIndex+1. This
108 // grain shares a ring-buffer position with the tailIndex grain.
109 auto writeGrainInfo = mxlGrainInfo{};
110 std::uint8_t* writeBuffer = nullptr;
111 {
112 REQUIRE(mxlFlowWriterOpenGrain(writer, headIndex + 1, &writeGrainInfo, &writeBuffer) == MXL_STATUS_OK);
113
114 INFO("The newly opened grain should identify headIndex+1.");
115 CAPTURE(writeGrainInfo.index, headIndex + 1, writeGrainInfo.validSlices, writeGrainInfo.totalSlices);
116 CHECK(writeGrainInfo.index == headIndex + 1);
117 }
118
119 // Sanity check: verify that headIndex has not yet moved.
120 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
121 REQUIRE(runtimeInfo.headIndex == headIndex);
122
123 // Reading a grain at tailIndex during an open write of
124 // headIndex+1 should return TOO_LATE to prevent reader/writer
125 // conflict during the writer's open/commit interval.
126 constexpr auto NOT_CHANGED_INDEX_SENTINEL_VALUE = std::numeric_limits<decltype(readGrainInfo.index)>::max();
127 readBuffer = nullptr;
128 readGrainInfo = {};
129 readGrainInfo.index = NOT_CHANGED_INDEX_SENTINEL_VALUE;
130 auto const grainReadStatusInsideWrite = mxlFlowReaderGetGrainNonBlocking(reader, tailIndex, &readGrainInfo, &readBuffer);
131 {
132 INFO("A tailIndex read should return TOO_LATE while a write of "
133 "headIndex+1 is open because both logical grains alias the "
134 "same physical ring-buffer position."
135
136 );
137 CHECK(grainReadStatusInsideWrite == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
138 }
139 {
140 INFO("Because the tailIndex grain read is expected to return TOO_LATE, "
141 "it should not return a payload or modify the grain index.");
142 CHECK(readBuffer == nullptr);
143 CHECK(readGrainInfo.index == NOT_CHANGED_INDEX_SENTINEL_VALUE);
144 }
145
146 // REVIEW-ONLY (pre-fix diagnostic): Retained so running this test
147 // against the old implementation also exposes the inconsistent
148 // state returned when MXL_STATUS_OK is returned.
149 if (grainReadStatusInsideWrite == MXL_STATUS_OK)
150 {
151 INFO("The tailIndex read was permitted, therefore the returned grain "
152 "metadata must be internally consistent: the payload must be "
153 "available, the grain index must identify the requested tailIndex "
154 "grain, and the full-grain read must report all slices as valid.");
155 CAPTURE(readBuffer, readGrainInfo.index, tailIndex, headIndex + 1, readGrainInfo.validSlices, readGrainInfo.totalSlices);
156 CHECK(readBuffer != nullptr);
157 CHECK(readGrainInfo.validSlices == readGrainInfo.totalSlices);
158
159 // REVIEW-ONLY: Fails with the pre-fix read boundary. The
160 // returned grain is headIndex+1 although tailIndex was
161 // requested.
162 CHECK(readGrainInfo.index == tailIndex);
163 }
164
165 // Reading tailIndex through the slice API during an open write of
166 // headIndex+1 should return TOO_LATE, should not return a payload,
167 // and should not modify readGrainInfo.validSlices.
168 constexpr auto NOT_CHANGED_SLICE_SENTINEL_VALUE = std::numeric_limits<decltype(readGrainInfo.validSlices)>::max();
169 readBuffer = nullptr;
170 readGrainInfo = {};
171 readGrainInfo.validSlices = NOT_CHANGED_SLICE_SENTINEL_VALUE;
172 auto const sliceReadStatus = mxlFlowReaderGetGrainSliceNonBlocking(reader, tailIndex, MXL_GRAIN_VALID_SLICES_ANY, &readGrainInfo, &readBuffer);
173 {
174 INFO("A tailIndex slice read should return TOO_LATE while a write of "
175 "headIndex+1 is open because both logical grains alias the "
176 "same physical ring-buffer position.");
177 CHECK(sliceReadStatus == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
178 }
179 {
180 INFO("Because the tailIndex slice read is expected to return TOO_LATE, "
181 "it should not return a payload or modify validSlices.");
182 CHECK(readBuffer == nullptr);
183 CHECK(readGrainInfo.validSlices == NOT_CHANGED_SLICE_SENTINEL_VALUE);
184 }
185
186 // REVIEW-ONLY (pre-fix diagnostic): Retained so running this test
187 // against the old implementation also exposes the inconsistent
188 // state returned when MXL_STATUS_OK is returned.
189 if (sliceReadStatus == MXL_STATUS_OK)
190 {
191 INFO("The tailIndex slice read was permitted, therefore the returned "
192 "grain state must be internally consistent: the payload must be "
193 "available, the grain index must identify the requested tailIndex "
194 "grain, and the previously committed tail grain must still be "
195 "reported as fully valid.");
196
197 CAPTURE(readBuffer, readGrainInfo.index, tailIndex, headIndex + 1, readGrainInfo.validSlices, readGrainInfo.totalSlices);
198 CHECK(readBuffer != nullptr);
199 CHECK(readGrainInfo.validSlices == readGrainInfo.totalSlices);
200
201 // REVIEW-ONLY: Fails with the pre-fix read boundary. The
202 // returned grain is headIndex+1 although tailIndex was
203 // requested.
204 CHECK(readGrainInfo.index == tailIndex);
205 }
206
207 // Commit the write and update local headIndex.
208 writeBuffer[0] = 0xCA;
209 writeBuffer[writeGrainInfo.grainSize - 1] = 0xFE;
210 writeGrainInfo.validSlices = writeGrainInfo.totalSlices;
211 REQUIRE(mxlFlowWriterCommitGrain(writer, &writeGrainInfo) == MXL_STATUS_OK);
212
213 headIndex = headIndex + 1;
214 tailIndex = headIndex - grainCount + 1;
215
216 // Sanity check local headIndex matches flow's headIndex.
217 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
218 REQUIRE(runtimeInfo.headIndex == headIndex);
219
220 // Post commit read of the tailIndex grain should continue to be
221 // TOO_LATE.
222 readBuffer = nullptr;
223 readGrainInfo = {};
224 auto const grainReadStatusPostCommit = mxlFlowReaderGetGrainNonBlocking(reader, tailIndex, &readGrainInfo, &readBuffer);
225 {
226 INFO("A post commit tailIndex read should return TOO_LATE after a "
227 "write of headIndex+1 is committed because both logical grains "
228 "alias the same physical ring-buffer position.");
229 CHECK(grainReadStatusPostCommit == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
230 }
231
232 // Tear down.

234
235 REQUIRE(mxlReleaseFlowWriter(instance, writer) == MXL_STATUS_OK);
236
237 REQUIRE(mxlDestroyInstance(instance) == MXL_STATUS_OK);
238}

240/*
241 This test verifies that the audio reader boundary between the
242 writer's half-ring and the reader's half-ring is enforced regardless
243 of whether the writer has a write operation open.
244
245 +--------------------------+--------------------------+--------------------------+--
246 | physical half 0 | physical half 1 | physical half 0 |
247 | writer's half-ring | reader's half-ring | writer's half-ring | ...
248 +--------------------------+--------------------------+--------------------------+--
249 ^ ^ ^
250 ringTailIndex boundaryIndex headIndex
251 ring wrap
252
253 ringTailIndex = headIndex - bufferLength + 1
254 boundaryIndex = headIndex - bufferLength/2 + 1
255
256 Reads at boundaryIndex are allowed. Reads at boundaryIndex-1 and
257 earlier are TOO_LATE.
258
259 Note: physical half 0/1 are used here to illustrate the two
260 half-ring regions and their reuse after ring wrap.
261*/
262TEST_CASE_PERSISTENT_FIXTURE(mxl::tests::mxlDomainFixture, "Audio Flow : tail read during open write", "[mxl flows]")
263{
264 auto const flowId = "b3bb5be7-9fe9-4324-a5bb-4c70e1084449";
265 auto const flowDef = mxl::tests::readFile("data/audio_flow.json");
266
267 // Init the domain instance.
268 auto instance = mxlCreateInstance(domain.string().c_str(), "");
269 REQUIRE(instance != nullptr);
270
271 // Init a flow writer.
272 auto writer = mxlFlowWriter{};
273 auto configInfo = mxlFlowConfigInfo{};
274 auto flowWasCreated = false;
275
276 REQUIRE(mxlCreateFlowWriter(instance, flowDef.c_str(), "", &writer, &configInfo, &flowWasCreated) == MXL_STATUS_OK);
277 REQUIRE(flowWasCreated);
278
279 // Init headIndex from the current MXL time.
280 auto const now = mxlGetTime();
281 auto headIndex = mxlTimestampToIndex(&configInfo.common.grainRate, now);
282 REQUIRE(headIndex != MXL_UNDEFINED_INDEX);
283
284 auto const bufferLength = static_cast<std::uint64_t>(configInfo.continuous.bufferLength);
285 auto const halfBufferLength = bufferLength / 2U;
286
287 REQUIRE(halfBufferLength > 0U);
288 REQUIRE(headIndex > bufferLength);
289
290 // Init the full ring buffer using two sequential half-buffer
291 // writes ending at headIndex. Each write respects the writer's
292 // half-ring limit.
293 {
294 auto const boundaryIndex = headIndex - halfBufferLength + 1;
295 auto tmpWriteIndex = boundaryIndex - 1;
296 for (auto i = std::uint64_t{0}; i < 2U; ++i)
297 {
298 auto writeBuffer = mxlMutableWrappedMultiBufferSlice{};
299
300 REQUIRE(mxlFlowWriterOpenSamples(writer, tmpWriteIndex, halfBufferLength, &writeBuffer) == MXL_STATUS_OK);
301
302 REQUIRE(mxlFlowWriterCommitSamples(writer) == MXL_STATUS_OK);
303
304 tmpWriteIndex += halfBufferLength;
305 }
306 }
307
308 // Init a reader and sanity check the flow's headIndex.
309 auto reader = mxlFlowReader{};
310 REQUIRE(mxlCreateFlowReader(instance, flowId, "", &reader) == MXL_STATUS_OK);
311
312 auto runtimeInfo = mxlFlowRuntimeInfo{};
313 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
314 REQUIRE(runtimeInfo.headIndex == headIndex);
315
316 // boundaryIndex separates the writer's half-ring from the reader's
317 // half-ring.
318 auto const boundaryIndex = headIndex - halfBufferLength + 1;
319 auto const ringTailIndex = headIndex - bufferLength + 1;
320
321 auto readBuffer = mxlWrappedMultiBufferSlice{};
322
323 // Reading at boundaryIndex is OK.
324 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, boundaryIndex, 1U, &readBuffer) == MXL_STATUS_OK);
325
326 // Reading one index before boundaryIndex is TOO_LATE.
327 readBuffer = {};
328 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, boundaryIndex - 1, 1U, &readBuffer) == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
329
330 // Open a write to the next sequential half-buffer at
331 // headIndex+halfBufferLength. This does *not* overlap the
332 // reader's half-ring.
333 auto writeBuffer = mxlMutableWrappedMultiBufferSlice{};
334 REQUIRE(mxlFlowWriterOpenSamples(writer, headIndex + halfBufferLength, halfBufferLength, &writeBuffer) == MXL_STATUS_OK);
335
336 // Sanity check: verify that headIndex has not yet moved.
337 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
338 REQUIRE(runtimeInfo.headIndex == headIndex);
339
340 // The reader boundary is unchanged while the write is open.
341 readBuffer = {};
342 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, boundaryIndex, 1U, &readBuffer) == MXL_STATUS_OK);
343
344 // A read at the tail is also in the writer's half-ring and should
345 // fail with TOO_LATE.
346 readBuffer = {};
347 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, ringTailIndex, 1U, &readBuffer) == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
348
349 // A read just before boundaryIndex should also fail with
350 // TOO_LATE.
351 readBuffer = {};
352 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, boundaryIndex - 1, 1U, &readBuffer) == MXL_ERR_OUT_OF_RANGE_TOO_LATE);
353
354 // Commit the write and update local headIndex.
355 REQUIRE(mxlFlowWriterCommitSamples(writer) == MXL_STATUS_OK);
356
357 headIndex = headIndex + halfBufferLength;
358
359 // Sanity check that the local headIndex matches the flow's
360 // headIndex.
361 REQUIRE(mxlFlowReaderGetRuntimeInfo(reader, &runtimeInfo) == MXL_STATUS_OK);
362 REQUIRE(runtimeInfo.headIndex == headIndex);
363
364 // Post commit read at the new boundaryIndex is OK.
365 auto const newBoundaryIndex = headIndex - halfBufferLength + 1;
366 readBuffer = {};
367
368 REQUIRE(mxlFlowReaderGetSamplesNonBlocking(reader, newBoundaryIndex, 1U, &readBuffer) == MXL_STATUS_OK);
369
370 // Tear down.
371 REQUIRE(mxlReleaseFlowReader(instance, reader) == MXL_STATUS_OK);
372
373 REQUIRE(mxlReleaseFlowWriter(instance, writer) == MXL_STATUS_OK);
374
375 REQUIRE(mxlDestroyInstance(instance) == MXL_STATUS_OK);
376}
