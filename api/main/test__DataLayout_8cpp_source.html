49TEST_CASE("ofi: DataLayout::Discrete planePayloadOffset", "[ofi][DataLayout]")
50{
51 constexpr auto headerSize = std::uint32_t{8192};
52
53 SECTION("single plane: plane 0 starts at the grain payload offset")
54 {
55 auto layout = DataLayout::Discrete{
56 .sliceSizes = {5120, 0, 0, 0},
57 .totalSlices = 1080
58 };
59 REQUIRE(layout.planePayloadOffset(0, headerSize) == headerSize);
60 }
61
62 SECTION("two planes: plane 0 at header, plane 1 after fill data")
63 {
64 constexpr auto height = std::uint16_t{1080};
65 constexpr auto fillSlice = std::uint32_t{5120};
66 constexpr auto keySlice = std::uint32_t{2560};
67
68 auto layout = DataLayout::Discrete{
69 .sliceSizes = {fillSlice, keySlice, 0, 0},
70 .totalSlices = height
71 };
72
73 REQUIRE(layout.planePayloadOffset(0, headerSize) == headerSize);
74 REQUIRE(layout.planePayloadOffset(1, headerSize) == headerSize + (height * fillSlice));
75 }
76
77 SECTION("four planes: each plane offset accumulates")
78 {
79 constexpr auto slices = std::uint32_t{10};
80 auto layout = DataLayout::Discrete{
81 .sliceSizes = {100, 200, 300, 400},
82 .totalSlices = slices
83 };
84
85 REQUIRE(layout.planePayloadOffset(0, headerSize) == headerSize);
86 REQUIRE(layout.planePayloadOffset(1, headerSize) == headerSize + (slices * 100));
87 REQUIRE(layout.planePayloadOffset(2, headerSize) == headerSize + (slices * 100) + (slices * 200));
88 REQUIRE(layout.planePayloadOffset(3, headerSize) == headerSize + (slices * 100) + (slices * 200) + (slices * 300));
89 }
90}
