141 template<typename Iter>
142 bool parse_object_item(picojson::input<Iter>& in, std::string const& key)
143 {
144 auto& object = _out->get<picojson::object>();
145 if (object.size() >= _limits->maxContainerSize)
146 {
147 *_error = fmt::format("JSON object has more than {} members.", _limits->maxContainerSize);
148 return false;
149 }
150 // picojson keeps the last of several equal keys. Reject them, because other parsers may keep the first.
151 auto const [it, inserted] = object.try_emplace(key);
152 if (!inserted)
153 {
154 *_error = fmt::format("JSON object has the key '{}' more than once.", key);
155 return false;
156 }
157 auto child = LimitedJsonParseContext{&it->second, *_limits, *_error, _depth + 1U};
158 return picojson::_parse(child, in);
159 }
160
161 private:
163 bool checkDepth()
164 {
165 if (_depth >= _limits->maxDepth)
166 {
167 *_error = fmt::format("JSON nesting is deeper than {} levels.", _limits->maxDepth);
168 return false;
169 }
170 return true;
171 }

193 inline std::string parseJson(picojson::value& out, std::string_view text, JsonLimits const& limits = {})
194 {
195 if (text.size() > limits.maxInputSize)
196 {
197 return fmt::format("JSON input has {} bytes, the limit is {}.", text.size(), limits.maxInputSize);
198 }
199
200 auto limitError = std::string{};
201 auto syntaxError = std::string{};
202 auto ctx = detail::LimitedJsonParseContext{&out, limits, limitError, 0U};
203 auto const end = picojson::_parse(ctx, text.begin(), text.end(), &syntaxError);
204 if (!limitError.empty())
205 {
206 return limitError;
207 }
208 if (!syntaxError.empty())
209 {
210 return syntaxError;
211 }
212
213 auto const isWhitespace = [](char c) noexcept
214 {
215 return (c == ' ') || (c == '\t') || (c == '\n') || (c == '\r');
216 };
217 if (!std::all_of(end, text.end(), isWhitespace))
218 {
219 return "Unexpected content after the JSON value.";
220 }
221 return {};
222 }

235 template<std::integral T>
236 inline T jsonToInteger(picojson::value const& value, std::string_view name)
237 {
238 if (!value.is<double>())
239 {
240 throw std::invalid_argument{fmt::format("'{}' must be a number.", name)};
241 }
242
243 auto const number = value.get<double>();
244
245 // 2^digits is the first value past the range of T. It is exact as a double, while numeric_limits<T>::max() may
246 // round up to it.
247 auto const upper = std::ldexp(1.0, std::numeric_limits<T>::digits);
248 auto const lower = std::numeric_limits<T>::is_signed ? -upper : 0.0;
249 if ((std::trunc(number) != number) || (number < lower) || (number >= upper))
250 {
251 throw std::invalid_argument{fmt::format(
252 "'{}' must be a whole number between {} and {}, got {}.", name, std::numeric_limits<T>::min(), std::numeric_limits<T>::max(), number)};
253 }
254 return static_cast<T>(number);
255 }
