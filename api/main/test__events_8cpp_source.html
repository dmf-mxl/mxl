244 std::uint32_t readNetworkWord(std::span<std::uint8_t const, 4> bytes) noexcept
245 {
246 // Widen each byte before shifting to decode safely without an aligned integer load.
247 return (std::uint32_t{bytes[0]} << 24) | (std::uint32_t{bytes[1]} << 16) | (std::uint32_t{bytes[2]} << 8) | std::uint32_t{bytes[3]};

265 void writeSt2110Packet(mxlFlowWriter writer, std::span<std::uint8_t const> rtp, std::uint64_t taiEpoch, std::uint32_t capacity)
266 {
267 REQUIRE(rtp.size() >= 20);
268 // Validate the supported RTP header layout, zero marker bit, and dynamic payload type.
269 REQUIRE(rtp[0] == 0x80);
270 REQUIRE(rtp[1] >= 96);
271 REQUIRE(rtp[1] <= 127);
272 auto const timestamp = readNetworkWord(rtp.subspan<4, 4>());
273 // Decode the 22-bit DIT, completion bit and 9-bit word count, which includes the offset word.
274 auto const itemHeader = readNetworkWord(rtp.subspan<12, 4>());
275 auto const lengthWords = itemHeader & 0x1FFU;
276 REQUIRE(lengthWords > 1);
277 REQUIRE(rtp.size() == 16 + (std::size_t{lengthWords} * 4));
278 auto const offsetWords = readNetworkWord(rtp.subspan<16, 4>());
279 REQUIRE(offsetWords <= std::numeric_limits<std::uint32_t>::max() / 4);
280 auto const data = rtp.subspan(20);
281
282 // Transport headers become event metadata. Only the segment bytes are copied into the ring.
283 auto info = mxlEventInfo{};
284 auto buffer = static_cast<std::uint8_t*>(nullptr);
285 REQUIRE(mxlFlowWriterOpenEvent(writer, &info, &buffer) == MXL_STATUS_OK);
286 REQUIRE(buffer != nullptr);
287 REQUIRE(data.size() <= capacity);
288 info.timestamp = taiEpoch + (std::uint64_t{timestamp} * 1'000'000'000 / 48'000);
289 info.registryType = MXL_EVENT_REGISTRY_TYPE_SMPTE;
290 auto const dit = fmt::format("{:X}", itemHeader >> 10);
291 REQUIRE(dit.size() < sizeof info.dataItemType);
292 std::memcpy(info.dataItemType, dit.c_str(), dit.size() + 1);
293 info.eventSize = static_cast<std::uint32_t>(data.size());
294 // ST 2110-41 offsets count 32-bit words; MXL offsets count bytes within the logical event.
295 info.offset = offsetWords * 4;
296 info.complete = static_cast<std::uint8_t>((itemHeader >> 9) & 1U);
297 std::memcpy(buffer, data.data(), data.size());
298 REQUIRE(mxlFlowWriterCommitEvent(writer, &info) == MXL_STATUS_OK);
299 }
