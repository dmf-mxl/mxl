17 std::uint64_t libfabricCaps(std::optional<ProviderCapabilities> const& capabilities, bool isTarget)
18 {
19 if (!capabilities)
20 {
21 return 0;
22 }
23 auto result = std::uint64_t{0};
24 if ((capabilities->interfaceCaps & MXL_FABRICS_IFACE_CAP_REMOTE_WRITE) != 0)
25 {
26 result |= (isTarget ? FI_REMOTE_WRITE : FI_WRITE) | FI_RMA;
27 }
28 if ((capabilities->interfaceCaps & MXL_FABRICS_IFACE_CAP_SEND_RECEIVE) != 0)
29 {
30 result |= FI_SEND | FI_RECV;
31 }
32 return result;
33 }

46 return result;

58 ProviderConfig ProviderConfig::create(Provider provider, bool isTarget, std::optional<ProviderCapabilities> capabilities)
59 {
60 switch (provider)
61 {
62 case Provider::TCP: return ProviderConfig::tcp(isTarget, capabilities);
63 case Provider::VERBS: return ProviderConfig::verbs(isTarget, capabilities);
64 case Provider::EFA: return ProviderConfig::efa(isTarget, capabilities);
65 case Provider::SHM: return ProviderConfig::shm(isTarget, capabilities);
66 default: throw Exception::invalidState("cannot create provider config for ANY provider");

135 bool ProviderConfig::isSupportedFabricInfo(FabricInfoView view) const noexcept
136 {
137 // Filters out all protocol types that are not supported for this provider.
138 auto const protocolNotSupported = std::ranges::find(_values.supportedProtocols, view->ep_attr->protocol) == _values.supportedProtocols.end();
139
140 // Filters out all address formats that are not supported for this provider.
141 auto const addressFormatNotSupported =
142 std::ranges::find(_values.supportedAddressFormats, view->addr_format) == _values.supportedAddressFormats.end();
143
144 // Filters out all info objects that have caps set that we can't support.
145 auto const containsFilteredCaps = ((view->caps & _values.filteredCaps) > 0);
146
147 // Filters out all info objects that are missing caps that we always need.
148 auto const missingRequiredCaps = (_values.requiredCaps != 0) && ((view->caps & _values.requiredCaps) != _values.requiredCaps);
149
150 // Filters out all objects that are not the endpoint type that we are looking for with this provider.
151 auto const unsupportedEndpointType = (view->ep_attr->type != _values.endpointType);
152
153 // Filters out all fabrics whose name is not in supportedFabricNames, if any are given.
154 auto const unsupportedFabricName =
155 !_values.supportedFabricNames.empty() &&
156 ((view->fabric_attr->name == nullptr) ||
157 std::ranges::find(_values.supportedFabricNames, std::string{view->fabric_attr->name}) == _values.supportedFabricNames.end());
158
159 return !(protocolNotSupported || addressFormatNotSupported || containsFilteredCaps || missingRequiredCaps || unsupportedEndpointType ||
160 unsupportedFabricName);
161 }
