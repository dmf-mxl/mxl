22 PosixEventFlowWriter::PosixEventFlowWriter(FlowManager const& manager, uuids::uuid const& flowId, std::unique_ptr<EventFlowData>&& data,
23 std::shared_ptr<DomainWatcher> const& watcher)
24 : EventFlowWriter{flowId, manager.getDomain()}
25 , _flowData{std::move(data)}
26 , _open{false}
27 , _lastTimestamp{0}
28 , _payload(_flowData->flowInfo()->config.event.eventPayloadSize)
29 , _watcher{watcher}
30 {
31 // The caller guarantees sole writer ownership, keeping the committed head
32 // stable while we restore the timestamp from an existing flow.
33 auto flow = _flowData->flow();
34 auto& runtime = flow->info.runtime;
35 auto head = EventRingBuffer::load(runtime.headIndex);
36 if (head != MXL_UNDEFINED_INDEX)
37 {
38 auto info = mxlEventInfo{};
39 if (_flowData->ring().read(head, info, _payload.data()) != EventRingBuffer::ReadResult::Ready)
40 {
41 throw std::runtime_error{"Invalid last committed event."};
42 }
43 _lastTimestamp = info.timestamp;
44 }
45 // A crashed writer may have published the next slot without advancing the head.
46 auto const next = (head == MXL_UNDEFINED_INDEX) ? 0 : head + 1;
47 auto info = mxlEventInfo{};
48 auto const result = _flowData->ring().read(next, info, _payload.data());
49 if (result == EventRingBuffer::ReadResult::Ready)
50 {
51 if (info.timestamp < _lastTimestamp)
52 {
53 throw std::runtime_error{"Invalid recovered event timestamp."};
54 }
55 _lastTimestamp = info.timestamp;
56 std::atomic_ref{runtime.lastWriteTime}.store(currentTime(Clock::TAI).value, std::memory_order_release);
57 std::atomic_ref{runtime.headIndex}.store(next, std::memory_order_release);
58 head = next;
59 }
60 else if (result != EventRingBuffer::ReadResult::Pending)
61 {
62 throw std::runtime_error{"Invalid next event during writer recovery."};
63 }
64 _watcher->addFlow(this, flowId);
65 if (head != MXL_UNDEFINED_INDEX)
66 {
67 // Also cover a crash after head advancement but before reader notification.
68 std::atomic_ref{flow->state.syncCounter}.fetch_add(1, std::memory_order_release);
69 wakeAll(&flow->state.syncCounter);
70 }
71 }

172 mxlStatus PosixEventFlowWriter::commit(mxlEventInfo const& info)
173 {
174 if (!_open || info.version != EVENT_HEADER_VERSION || info.size != sizeof info || info.flags != 0 || info.timestamp < _lastTimestamp ||
175 info.eventSize > _flowData->flowInfo()->config.event.eventPayloadSize)
176 {
177 return MXL_ERR_INVALID_ARG;
178 }
179 auto flow = _flowData->flow();
180 auto& runtime = flow->info.runtime;
181 auto const head = EventRingBuffer::load(runtime.headIndex);
182 auto const index = head == MXL_UNDEFINED_INDEX ? 0 : head + 1;
183 auto const status = _flowData->ring().publish(index, info, _payload.data());
184 if (status != MXL_STATUS_OK)
185 {
186 return status;
187 }
188 std::atomic_ref{runtime.lastWriteTime}.store(currentTime(Clock::TAI).value, std::memory_order_release);
189 std::atomic_ref{runtime.headIndex}.store(index, std::memory_order_release);
190 _lastTimestamp = info.timestamp;
191 _open = false;
192 std::atomic_ref{flow->state.syncCounter}.fetch_add(1, std::memory_order_release);
193 wakeAll(&flow->state.syncCounter);
194 return MXL_STATUS_OK;
195 }
